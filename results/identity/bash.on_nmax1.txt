#!/usr/bin/env bash

# watch_dir.sh - Watch a directory and print new files as they appear

set -euo pipefail

# Default to current directory if no argument provided
WATCH_DIR="${1:-.}"

# Validate directory exists
if [[ ! -d "$WATCH_DIR" ]]; then
    echo "Error: '$WATCH_DIR' is not a valid directory." >&2
    exit 1
fi

# Resolve to absolute path
WATCH_DIR="$(cd "$WATCH_DIR" && pwd)"

echo "Watching: $WATCH_DIR"
echo "Press Ctrl+C to stop."
echo "--------------------------------"

    # inotifywait approach (efficient, event-driven)

            # Only print regular files (skip directories)

                echo "New file: $file"
            fi
